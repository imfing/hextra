{{- /* Extracts all headings from a page and adds them to the scratchpad. The keys can be obtained from the scratchpad by using the "keys" key. The opening heading markers and closing tags can be obtained from the scratchpad by using the "markers" and "endTags" keys. The heading IDs can be obtained from the scratchpad by using the "ids" key. The scratchpad must be initialized with empty slices before calling this function for the keys "keys", "markers", "endTags", and "ids". @param {any} target The element to extract headings from. @param {any} scratch The scratchpad to add the heading metadata to. @example {{ partial "utils/extract-headings.html" (dict "target" $h1 "scratch" $s) }} */ -}} {{- range $heading := index .target.Headings -}} {{- if and (eq $heading.Level 0) (not $heading.Title) -}} {{- $.scratch.Add "keys" (slice $heading.Title) -}} {{- $.scratch.Add "markers" (slice "") -}} {{- $.scratch.Add "endTags" (slice "") -}} {{- $.scratch.Add "ids" (slice "") -}} {{- else -}} {{- $headingTitle := $heading.Title | plainify | htmlUnescape | strings.TrimSpace -}} {{- $key := (printf "%s#%s" $heading.ID $headingTitle) -}} {{- $.scratch.Add "keys" (slice $key) -}} {{- $marker := printf "" $heading.Level)) -}} {{- $.scratch.Add "ids" (slice $heading.ID) -}} {{- end -}} {{- partial "utils/extract-headings.html" (dict "target" $heading "scratch" $.scratch ) }} {{- end -}}
