{{- /* OpenAPI (Swagger UI) */ -}} {{- $openapiBase := "" -}} {{- $useDefaultCdn := true -}} {{- with site.Params.openapi.base -}} {{- $openapiBase = . -}} {{- $useDefaultCdn = false -}} {{- end -}} {{- $openapiJsAsset := "" -}} {{- with site.Params.openapi.js -}} {{- $openapiJsAsset = . -}} {{- end -}} {{- $openapiCssAsset := "" -}} {{- with site.Params.openapi.css -}} {{- $openapiCssAsset = . -}} {{- end -}} {{- /* If only js/css is set without base, use local asset loading */ -}} {{- if and $useDefaultCdn (or (ne $openapiJsAsset "") (ne $openapiCssAsset "")) -}} {{- $useDefaultCdn = false -}} {{- end -}} {{- /* Set default CDN base if needed */ -}} {{- if $useDefaultCdn -}} {{- $openapiBase = "https://cdn.jsdelivr.net/npm/swagger-ui-dist@5" -}} {{- end -}} {{- $isRemoteBase := or (strings.HasPrefix $openapiBase "http://") (strings.HasPrefix $openapiBase "https://") -}} {{- /* CSS retrieval: get raw CSS from either local asset or remote, then process */ -}} {{- if $isRemoteBase -}} {{- $cssPath := cond (ne $openapiCssAsset "") $openapiCssAsset "swagger-ui.css" -}} {{- $openapiCssUrl := urls.JoinPath $openapiBase $cssPath -}} {{- with try (resources.GetRemote $openapiCssUrl) -}} {{- with .Err -}} {{- errorf "Could not retrieve Swagger UI css file from %s. Reason: %s." $openapiCssUrl . -}} {{- else with .Value -}} {{- with resources.Copy "css/swagger-ui.css" . -}} {{- $openapiCss := . | fingerprint -}} {{- end -}} {{- else -}} {{- errorf "Could not retrieve Swagger UI css file from %s." $openapiCssUrl -}} {{- end -}} {{- end -}} {{- else if $openapiCssAsset -}} {{- with resources.Get $openapiCssAsset -}} {{- $openapiCss := . | fingerprint -}} {{- else -}} {{- errorf "Swagger UI css asset not found at %q" $openapiCssAsset -}} {{- end -}} {{- end -}} {{- /* JS retrieval: get raw JS from either local asset or remote, then process */ -}} {{- if $isRemoteBase -}} {{- $jsPath := cond (ne $openapiJsAsset "") $openapiJsAsset "swagger-ui-bundle.js" -}} {{- $openapiJsUrl := urls.JoinPath $openapiBase $jsPath -}} {{- with try (resources.GetRemote $openapiJsUrl) -}} {{- with .Err -}} {{- errorf "Could not retrieve Swagger UI js file from %s. Reason: %s." $openapiJsUrl . -}} {{- else with .Value -}} {{- with resources.Copy "js/swagger-ui-bundle.js" . -}} {{- $openapiJs := . | fingerprint -}} {{- end -}} {{- else -}} {{- errorf "Could not retrieve Swagger UI js file from %s." $openapiJsUrl -}} {{- end -}} {{- end -}} {{- else if $openapiJsAsset -}} {{- with resources.Get $openapiJsAsset -}} {{- $openapiJs := . | fingerprint -}} {{- else -}} {{- errorf "Swagger UI js asset not found at %q" $openapiJsAsset -}} {{- end -}} {{- end -}}
