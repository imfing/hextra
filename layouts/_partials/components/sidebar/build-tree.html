{{- $context := .context -}} {{- $page := .page -}} {{- $pageURL := .pageURL -}} {{- $matched := .matched | default false -}} {{- $autoNodes := partialCached "components/sidebar/build-auto-nodes.html" (dict "context" $context) $context.RelPermalink -}} {{- $nodes := $autoNodes -}} {{- if $matched -}} {{- $sidebarData := partial "components/sidebar/source-data.html" $page -}} {{- if gt (len $sidebarData) 0 -}} {{- $dataNodes := partialCached "components/sidebar/build-data-tree.html" (dict "nodes" $sidebarData) $page.Language.Lang -}} {{- /* Keep only data roots that live under $context */ -}} {{- $scopedData := slice -}} {{- range $dataNodes -}} {{- $node := . -}} {{- with $node.page -}} {{- if or (eq .RelPermalink $context.RelPermalink) (.IsDescendant $context) -}} {{- $scopedData = $scopedData | append $node -}} {{- end -}} {{- end -}} {{- end -}} {{- if gt (len $scopedData) 0 -}} {{- $dataByLink := dict -}} {{- $autoClaimed := dict -}} {{- range $scopedData -}} {{- if .link -}} {{- $dataByLink = merge $dataByLink (dict .link .) -}} {{- end -}} {{- end -}} {{- $nodes = slice -}} {{- range $autoNodes -}} {{- $autoNode := . -}} {{- with index $dataByLink $autoNode.link -}} {{- $nodes = $nodes | append . -}} {{- else -}} {{- $nodes = $nodes | append $autoNode -}} {{- end -}} {{- $autoClaimed = merge $autoClaimed (dict $autoNode.link true) -}} {{- end -}} {{- range $scopedData -}} {{- if and .link (not (index $autoClaimed .link)) -}} {{- $nodes = $nodes | append . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $nodes -}}
