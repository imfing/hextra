{{- $context := .context -}} {{- $level := .level | default 0 -}} {{- $scope := .scope | default "sidebar" -}} {{- $allowSeparator := false -}} {{- if isset . "allowSeparator" -}} {{- $allowSeparator = .allowSeparator -}} {{- else if eq $level 0 -}} {{- $allowSeparator = true -}} {{- end -}} {{- $nodes := slice -}} {{- if lt $level (site.Params.sidebar.maxDepth | default 4) -}} {{- $sorted := partial "components/sidebar/sorted-items.html" (dict "context" $context "scope" $scope) -}} {{- range $sorted -}} {{- $sep := .Params.sidebar.separator -}} {{- $sepLift := and (eq $sep true) $allowSeparator -}} {{- if and (eq $sep true) (not $allowSeparator) -}} {{- warnf "sidebar.separator: true on %q is only respected at the top level (depth %d); ignored" .RelPermalink $level -}} {{- end -}} {{- if $sepLift -}} {{- /* Section itself renders as a separator; its children are lifted. */ -}} {{- $nodes = $nodes | append (dict "link" "" "title" (partial "utils/title" .) "type" "separator" "items" slice "icon" (.Params.sidebar.icon | default "") ) -}} {{- if or .RegularPages .Sections -}} {{- $lifted := partialCached "components/sidebar/build-auto-nodes.html" (dict "context" . "level" $level "scope" $scope "allowSeparator" false) .RelPermalink $level $scope false -}} {{- $nodes = $nodes | append $lifted -}} {{- end -}} {{- else -}} {{- $children := slice -}} {{- if or .RegularPages .Sections -}} {{- $children = partialCached "components/sidebar/build-auto-nodes.html" (dict "context" . "level" (add $level 1) "scope" $scope "allowSeparator" false) .RelPermalink (add $level 1) $scope false -}} {{- end -}} {{- $nodes = $nodes | append (dict "link" .RelPermalink "title" (partial "utils/title" .) "items" $children "open" .Params.sidebar.open "icon" (.Params.sidebar.icon | default "") ) -}} {{- end -}} {{- end -}} {{- end -}} {{- return $nodes -}}
