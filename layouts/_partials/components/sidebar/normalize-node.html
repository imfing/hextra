{{- $node := . -}} {{- $link := $node.link | default "" -}} {{- $external := hasPrefix $link "http" -}} {{- $normalizedLink := $link -}} {{- $page := false -}} {{- if and $link (not $external) -}} {{- $normalizedLink = partial "utils/sidebar-link.html" $link -}} {{- with site.GetPage $link -}} {{- $page = . -}} {{- end -}} {{- end -}} {{- $title := $node.title -}} {{- if and (not $title) $page -}} {{- $title = partial "utils/title" $page -}} {{- end -}} {{- $type := $node.type | default "" -}} {{- if and $type (not (in (slice "search" "separator") $type)) -}} {{- warnf "sidebar data: unknown type %q on node with link %q" $type ($node.link | default "") -}} {{- $type = "" -}} {{- end -}} {{- $knownKeys := slice "link" "title" "items" "merge" "open" "weight" "exclude" "icon" "type" -}} {{- range $key, $_ := $node -}} {{- if not (in $knownKeys $key) -}} {{- warnf "sidebar data: unknown key %q on node with link %q" $key ($node.link | default "") -}} {{- end -}} {{- end -}} {{- $icon := $node.icon | default "" -}} {{- if and (not $icon) $page -}} {{- $icon = $page.Params.sidebar.icon | default "" -}} {{- end -}} {{- $result := dict "type" $type "link" $normalizedLink "title" ($title | default "Untitled") "items" ($node.items | default slice) "merge" ($node.merge | default "none") "open" $node.open "weight" ($node.weight | default 0) "external" $external "page" $page "icon" $icon -}} {{- if eq $type "separator" -}} {{- $result = merge $result (dict "separator" true) -}} {{- end -}} {{- return $result -}}
